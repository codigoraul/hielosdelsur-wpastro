DirectoryIndex index.html
Options -Indexes
ErrorDocument 404 /404.html

<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  # Forzar HTTPS y sin www
  RewriteCond %{HTTPS} off [OR]
  RewriteCond %{HTTP_HOST} ^www\. [NC]
  RewriteRule ^(.*)$ https://hielosdelsur.cl/$1 [R=301,L]

  # No tocar el WordPress headless
  RewriteRule ^admin(/.*)?$ - [L]

  # Astro genera carpetas con index.html: /producto/silla-tita/ -> existe.
  # Si piden sin barra final y existe la carpeta, redirigir con barra (una sola URL canónica)
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteCond %{REQUEST_URI} !/$
  RewriteRule ^(.*)$ /$1/ [R=301,L]
</IfModule>
